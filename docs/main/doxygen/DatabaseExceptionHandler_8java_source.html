57import java.sql.SQLException;
58import java.util.UUID;

60@RestControllerAdvice
61@Order(1)
62public class DatabaseExceptionHandler {
63
64 private static final Logger LOGGER = LoggerFactory.getLogger(DatabaseExceptionHandler.class);
65
66 MariaDBErrorResolver resolver;
67
68 public DatabaseExceptionHandler(MariaDBErrorResolver resolver) {
69 this.resolver = resolver;
70 }
71
72 @ExceptionHandler(SQLException.class)
73 public ResponseEntity<String> handleSQLException(SQLException ex) {
74 UUID uuid = UUID.randomUUID();
75 JsonObjectBuilder errBuilder = Json.createObjectBuilder();
76 MariaDBError mariaDBError = resolver.resolve(ex.getErrorCode());
77 LOGGER.error("Error ID and message {} , {}", uuid, mariaDBError.message(), ex);
78 errBuilder.add("message", mariaDBError.message());
79 errBuilder.add("UUID", uuid.toString());
80 return new ResponseEntity<>(errBuilder.build().toString(), mariaDBError.status());
81
82 }
